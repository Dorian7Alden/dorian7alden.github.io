        // 训练数据
        const workoutData = {
            "10-01": 20, "10-02": 23, "10-04": 18, "10-05": 28, "10-08": 24, "10-09": 30,
            "10-11": 36, "10-14": 28, "10-21": 23, "10-25": 28, "10-27": 26, "10-28": 10,
            "11-01": 32, "11-02": 35, "11-03": 7, "11-04": 22, "11-05": 37, "11-07": 5,
            "11-10": 41, "11-11": 34, "11-14": 25, "11-20": 35, "11-28": 40, "11-29": 44
        };

        // 格子颜色逻辑
        function getCellColor(count) {
            if (count === 0) return "bg-gray-200";
            if (count <= 15) return "bg-green-200";
            if (count <= 30) return "bg-green-300";
            return "bg-green-500";
        }

        // 生成月份打卡墙
        function renderMonthGrid(month, daysInMonth, gridId) {
            const grid = document.getElementById(gridId);
            grid.innerHTML = '';

            // 2025年10月1日是周三（0=周日），11月1日是周六
            const firstDayOfWeek = month === 10 ? 3 : 6;
            const monthPrefix = month.toString().padStart(2, '0');

            // 补第一周空白格
            for (let i = 0; i < firstDayOfWeek; i++) {
                const emptyCell = document.createElement("div");
                emptyCell.className = "contribution-cell bg-gray-200";
                grid.appendChild(emptyCell);
            }

            // 填充日期格子
            for (let day = 1; day <= daysInMonth; day++) {
                const dayStr = day.toString().padStart(2, '0');
                const dateKey = `${monthPrefix}-${dayStr}`;
                const count = workoutData[dateKey] || 0;

                const cell = document.createElement("div");
                cell.className = `contribution-cell ${getCellColor(count)}`;
                cell.title = `${month}月${day}日: ${count > 0 ? count + '个' : '未训练'}`;
                grid.appendChild(cell);
            }
        }

        // 渲染打卡墙
        renderMonthGrid(10, 31, "october-grid");
        renderMonthGrid(11, 30, "november-grid");

        // 趋势线计算
        const calculateTrend = (data) => {
            const dates = Object.keys(data);
            const counts = Object.values(data);
            const sumX = dates.reduce((sum, _, index) => sum + index, 0);
            const sumY = counts.reduce((sum, count) => sum + count, 0);
            const sumXY = dates.reduce((sum, _, index) => sum + (index * counts[index]), 0);
            const sumX2 = dates.reduce((sum, _, index) => sum + (index * index), 0);
            const n = dates.length;

            const slope = (n * sumXY - sumX * sumY) / (n * sumX2 - sumX * sumX);
            const intercept = (sumY - slope * sumX) / n;

            return dates.map((_, index) => slope * index + intercept);
        };

        const trendData = calculateTrend(workoutData);
